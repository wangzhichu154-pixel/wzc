cmake_minimum_required(VERSION 3.20)
project(trading_engine VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENGINE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_library(trading_core src/itch.cpp src/book.cpp src/risk.cpp src/sequencer.cpp src/strategy.cpp src/simulation.cpp)
target_include_directories(trading_core PUBLIC include)
set_target_properties(trading_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(trading_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ENGINE_SANITIZERS)
  target_compile_options(trading_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(trading_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(trading-replay src/replay.cpp)
target_link_libraries(trading-replay PRIVATE trading_core)
add_executable(trading-ml-replay src/replay.cpp src/ml_io.cpp)
target_compile_definitions(trading-ml-replay PRIVATE TRADING_ML_CLI=1)
target_link_libraries(trading-ml-replay PRIVATE trading_core)
target_compile_options(trading-ml-replay PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(vector_core SHARED src/vector_core.cpp)
target_link_libraries(vector_core PRIVATE trading_core)
target_include_directories(vector_core PUBLIC include)
target_compile_options(vector_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ENGINE_SANITIZERS)
  # Consumers of an instrumented shared library need the runtime first too.
  target_compile_options(vector_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vector_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(engine-tests tests/test_engine.cpp)
target_link_libraries(engine-tests PRIVATE trading_core)
enable_testing()
add_test(NAME engine-tests COMMAND engine-tests)
find_package(Threads REQUIRED)
add_executable(cabi-tests tests/test_cabi.cpp)
target_link_libraries(cabi-tests PRIVATE vector_core Threads::Threads)
add_test(NAME cabi-tests COMMAND cabi-tests)
add_executable(c-header-test tests/test_c_header.c)
target_link_libraries(c-header-test PRIVATE vector_core)
add_test(NAME c-header-test COMMAND c-header-test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME replay-cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py)
  set_tests_properties(replay-cli-tests PROPERTIES ENVIRONMENT "TRADING_REPLAY_BIN=$<TARGET_FILE:trading-replay>")
  add_test(NAME ml-replay-cli-tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ml_cli.py)
  set_tests_properties(ml-replay-cli-tests PROPERTIES ENVIRONMENT "TRADING_ML_REPLAY_BIN=$<TARGET_FILE:trading-ml-replay>;TRADING_REPLAY_BIN=$<TARGET_FILE:trading-replay>")
endif()
install(TARGETS trading-replay RUNTIME DESTINATION bin)
install(TARGETS trading-ml-replay RUNTIME DESTINATION bin)
install(TARGETS vector_core LIBRARY DESTINATION lib)
install(FILES include/trading/vector_core.h DESTINATION include/trading)
